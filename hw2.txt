let rec makeList n = 
  match n with
  | _ when n < 0 -> failwith "Negative number" (* n is negative exception is thrown *)
  | 0 -> [] (* zero return empty *)
  | _ -> makeList (n - 1) @ [n];; (* cycle through decrementing n and concatinating with n*)
makeList 5;;

let makeListTail n = 
  let rec helper x lst = 
    match x with
    | _ when x < 0 -> failwith "Negative number" (* n is negative exception is thrown *)
    | 0 -> lst (* When x equals 0 return our list *)
    | _ -> helper (x - 1) (x::lst) (* Decrement x and add it to our list*)
  in
  helper n [];; (* starting from n with empty list *)
makeListTail 5;;

(* Derrick Pavia
   Problem 4 *)
type 'a btree =
  Empty
  | Leaf of 'a
  | Node of ( 'a * 'a btree * 'a btree) ;;
let tst = Node( 5, Leaf 2, Node( 12, Node(8, Leaf 7, Leaf 10), Leaf 49 ) ) ;;

let rec numLeaves tree =
  match tree with
  | Empty -> 0 (* No leafs return 0 *)
  | Leaf _ -> 1 (* 1 Leaf return 1 *)
  | Node (_, left, right) -> 
    numLeaves left + numLeaves right;; (* Recurse down left and right nodes counting and adding number of leaves*)
numLeaves tst;;

(* Derrick Pavia
   Problem 6 *)
let rec numNodes tree = 
  match tree with
  | Empty -> [] (* Empty return empty*)
  | Leaf x -> [x] (* Leaf Node returns its value *)
  | Node(x, left, right) -> 
    x :: (numNodes left @ numNodes right);; (* Add values to list in preorder *)

numNodes tst;;

divisors :: Int -> [Int] 
   divisors n = [x| x <- [1..n], n `rem` x == 0]
